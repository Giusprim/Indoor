// ═══════════════════════════════════════════ // IMPORTAZIONE E VALIDAZIONE (AGGIORNATO) // ═══════════════════════════════════════════ let TRACKS = []; function importManual(){ const raw = document.getElementById("in-man").value.trim(); if(!raw){ setImportStatus("Inserisci almeno un brano.","err"); return; } TRACKS = []; const lines = raw.split(/\n/).filter(x => x.trim()); let count = 0; lines.forEach((line, index) => { // Supporta sia | che il parsing intelligente dei dati const parts = line.split("|").map(x => x.trim()); if(parts.length < 3) { console.warn(`Riga ${index + 1} incompleta: attesi 3 campi (Titolo|BPM|Durata)`); return; } const info = parts[0]; // Titolo — Artista const bpm = parseInt(parts[1]); const duration = parseTime(parts[2]); // Validazione Campi Minimi if(!info || isNaN(bpm) || duration <= 0) { console.error(`Dati invalidi alla riga ${index + 1}`); return; } let title = info, artist = "Unknown"; // Gestione di diversi tipi di trattini (corto e lungo) if(info.includes("—")) { [title, artist] = info.split("—").map(x => x.trim()); } else if(info.includes("-")) { [title, artist] = info.split("-").map(x => x.trim()); } TRACKS.push({ i: ++count, title: title, ar: artist, bpm: String(bpm), d: duration, z: "—", rpm: "—", res: "—", rn: 0, tec: "—", mani: "P2", fase: "—", focus: "—", s: 0 }); }); if(TRACKS.length === 0){ setImportStatus("Errore: nessun brano valido trovato. Controlla il formato Titolo|BPM|Durata", "err"); } else { calcStartTimes(); renderCards(); setImportStatus(`✓ ${TRACKS.length} brani pronti per l'analisi.`, "ok"); } } // ═══════════════════════════════════════════ // ANALISI AI (ENDPOINT v1 STABILE) // ═══════════════════════════════════════════ async function runAI(){ const apiKey = loadApiKey(); if(!apiKey || TRACKS.length === 0) return; const btn = document.getElementById("btn-ai"); const logBox = document.getElementById("ai-log"); btn.disabled = true; btn.textContent = "⌛ Analisi in corso..."; clearLog(); log("→ Avvio analisi su endpoint stabile v1..."); const protocol = document.getElementById("wk-protocol").value; const prompt = `Agisci come Master Instructor Indoor Cycling. Genera profilo per: ${protocol}. Playlist: ${TRACKS.map(t => `${t.i}. ${t.title} | BPM ${t.bpm} | ${fmt(t.d)}`).join("\n")} Rispondi SOLO con array JSON: [{"zona":"Z2","rpm_label":"90","res_label":"35%","res_num":0.35,"tecnica":"Seduto","mani":"P2","fase":"Lavoro","focus":"Mantieni"}]`; try { // Cambiato endpoint a v1 per evitare Status 400 const url = `https://generativelanguage.googleapis.com/v1/models/gemini-1.5-flash:generateContent?key=${apiKey}`; const response = await fetch(url, { method: "POST", headers: {"Content-Type": "application/json"}, body: JSON.stringify({ contents: [{parts: [{text: prompt}]}], generationConfig: { responseMimeType: "application/json", temperature: 0.2 } }) }); if(!response.ok) throw new Error(`Status ${response.status}`); const data = await response.json(); const rawText = data.candidates[0].content.parts[0].text; const results = JSON.parse(rawText); TRACKS = TRACKS.map((t, i) => ({...t, ...results[i]})); renderCards(); log("✓ Analisi completata con successo."); } catch (e) { log(`✗ Errore: ${e.message}. Verifica la validità della API Key.`); } finally { btn.disabled = false; btn.textContent = "🤖 Analizza con Gemini AI"; } }
